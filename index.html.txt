    <!-- CANVAS AREA -->
    <div id="canvas-container" class="w-full h-[55vh] md:h-full md:flex-1 relative bg-gray-900 border-b md:border-b-0 md:border-r border-gray-700">
        <select id="up-axis-select" class="absolute top-2 right-2 bg-gray-800 text-gray-100 text-xs border border-gray-600 rounded p-1 outline-none shadow z-10">
            <option value="Z">Z-Alto (Default)</option>
            <option value="Y">Y-Alto</option>
            <option value="X">X-Alto</option>
        </select>
    </div>

    <!-- CONTROLS AREA -->
    <div class="w-full h-[45vh] md:h-full md:w-80 flex flex-col bg-gray-800 shadow-2xl z-20">
        <div id="joints-container" class="flex-1 overflow-y-auto p-2 space-y-2">
            <!-- Cards iniettate via JS -->
        </div>
        <div class="p-2 bg-gray-900 border-t border-gray-700">
            <button id="add-joint-btn" class="w-full py-2 bg-blue-600 hover:bg-blue-500 active:bg-blue-700 text-white text-sm font-semibold rounded transition-colors shadow">
                + Aggiungi Giunto
            </button>
        </div>
    </div>

    <script>
        // --- STATO DEL ROBOT ---
        let dhParams = [
            { type: 'R', a: 0, alpha: 0, d: 2, theta: 0 },
            { type: 'R', a: 3, alpha: 0, d: 0, theta: 45 },
            { type: 'P', a: 0, alpha: 90, d: 3, theta: 0 }
        ];

        // --- SETUP THREE.JS ---
        const container = document.getElementById('canvas-container');
        const scene = new THREE.Scene();
        scene.background = new THREE.Color(0x111827);

        const camera = new THREE.PerspectiveCamera(50, container.clientWidth / container.clientHeight, 0.1, 1000);
        // RIGOROSO: L'asse Y è SEMPRE l'alto del mondo Three.js. L'orbita sarà sempre corretta.
        camera.up.set(0, 1, 0); 
        camera.position.set(10, 10, 10);

        const renderer = new THREE.WebGLRenderer({ antialias: true });
        renderer.setSize(container.clientWidth, container.clientHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
        container.appendChild(renderer.domElement);

        const controls = new THREE.OrbitControls(camera, renderer.domElement);
        controls.enableDamping = true;
        controls.dampingFactor = 0.05;

        // Luci
        scene.add(new THREE.AmbientLight(0xffffff, 0.5));
        const dLight1 = new THREE.DirectionalLight(0xffffff, 0.8);
        dLight1.position.set(10, 20, 10);
        scene.add(dLight1);
        const dLight2 = new THREE.DirectionalLight(0xffffff, 0.3);
        dLight2.position.set(-10, -10, -10);
        scene.add(dLight2);

        // Griglia sul piano XZ (Y-Up nativo)
        const gridHelper = new THREE.GridHelper(20, 20, 0x374151, 0x1f2937);
        scene.add(gridHelper);

        const robotGroup = new THREE.Group();
        scene.add(robotGroup);

        // Materiali per Giunti e Link
        const jointGeom = new THREE.SphereGeometry(0.35, 32, 32);
        const jointMat = new THREE.MeshStandardMaterial({ color: 0x9ca3af, metalness: 0.8, roughness: 0.2 });
        const linkMat = new THREE.MeshStandardMaterial({ color: 0x6b7280, metalness: 0.3, roughness: 0.7 });

        // --- FUNZIONI DI MAPPATURA E CALCOLO ---

        // Questa funzione mappa lo spazio DH (Z=alto, X=avanti) nello spazio visivo ThreeJS (Y=alto)
        // Così il robot è sempre dritto, e i controlli camera non si rompono mai.
        function mapDHtoThree(vecDH) {
            return new THREE.Vector3(vecDH.x, vecDH.z, -vecDH.y);
        }

        // Determina i colori degli assi in base al menu a tendina
        // Non ruota il mondo, cambia solo "l'etichetta visiva" (colore) per mostrare qual è l'asse
        function getAxisColors() {
            const upAxis = document.getElementById('up-axis-select').value;
            // Restituisce { x, y, z } colori relativi agli assi DH
            if (upAxis === 'Z') return { x: 0xff0000, y: 0x00ff00, z: 0x0000ff }; // Z blu è verticale
            if (upAxis === 'Y') return { x: 0x0000ff, y: 0xff0000, z: 0x00ff00 }; // Y verde è verticale, Z blu orizzontale
            if (upAxis === 'X') return { x: 0x00ff00, y: 0x0000ff, z: 0xff0000 }; // X rosso è verticale
            return { x: 0xff0000, y: 0x00ff00, z: 0x0000ff };
        }

        function getDHMatrix(theta, d, a, alpha) {
            const th = THREE.MathUtils.degToRad(theta);
            const al = THREE.MathUtils.degToRad(alpha);
            const ct = Math.cos(th), st = Math.sin(th);
            const ca = Math.cos(al), sa = Math.sin(al);
            
            const m = new THREE.Matrix4();
            m.set(
                ct, -st*ca,  st*sa, a*ct,
                st,  ct*ca, -ct*sa, a*st,
                0,   sa,     ca,    d,
                0,   0,      0,     1
            );
            return m;
        }

        function drawReferenceFrame(origin, dirX, dirY, dirZ, colors) {
            const length = 1.2;
            const headLength = 0.3;
            const headWidth = 0.2;
            robotGroup.add(new THREE.ArrowHelper(dirX, origin, length, colors.x, headLength, headWidth));
            robotGroup.add(new THREE.ArrowHelper(dirY, origin, length, colors.y, headLength, headWidth));
            robotGroup.add(new THREE.ArrowHelper(dirZ, origin, length, colors.z, headLength, headWidth));
        }

        function drawLink(startPos, endPos) {
            const dist = startPos.distanceTo(endPos);
            if (dist < 0.001) return;

            const cylGeom = new THREE.CylinderGeometry(0.15, 0.15, dist, 16);
            const mesh = new THREE.Mesh(cylGeom, linkMat);
            
            // Posiziona a metà strada
            mesh.position.copy(startPos).add(endPos).multiplyScalar(0.5);
            
            // Orienta
            const dir = new THREE.Vector3().copy(endPos).sub(startPos).normalize();
            const up = new THREE.Vector3(0, 1, 0);
            const quaternion = new THREE.Quaternion().setFromUnitVectors(up, dir);
            mesh.setRotationFromQuaternion(quaternion);
            
            robotGroup.add(mesh);
        }

        function buildRobot() {
            // Svuota la scena
            while(robotGroup.children.length > 0){ robotGroup.remove(robotGroup.children[0]); }

            const colors = getAxisColors();
            let currentMatrix = new THREE.Matrix4(); 
            let prevThreePos = mapDHtoThree(new THREE.Vector3(0,0,0));

            // Disegna Frame Base e Giunto Base
            drawReferenceFrame(
                prevThreePos,
                mapDHtoThree(new THREE.Vector3(1,0,0)),
                mapDHtoThree(new THREE.Vector3(0,1,0)),
                mapDHtoThree(new THREE.Vector3(0,0,1)),
                colors
            );
            const baseSphere = new THREE.Mesh(jointGeom, jointMat);
            baseSphere.position.copy(prevThreePos);
            robotGroup.add(baseSphere);

            dhParams.forEach((param) => {
                const T_i = getDHMatrix(param.theta, param.d, param.a, param.alpha);
                currentMatrix.multiply(T_i);

                // Calcola assi globali nello spazio DH
                const originDH = new THREE.Vector3(0,0,0).applyMatrix4(currentMatrix);
                const dirXDH = new THREE.Vector3(1,0,0).applyMatrix4(currentMatrix).sub(originDH).normalize();
                const dirYDH = new THREE.Vector3(0,1,0).applyMatrix4(currentMatrix).sub(originDH).normalize();
                const dirZDH = new THREE.Vector3(0,0,1).applyMatrix4(currentMatrix).sub(originDH).normalize();

                // Mappa su spazio visivo Three.js
                const mappedOrigin = mapDHtoThree(originDH);
                const mappedDirX = mapDHtoThree(dirXDH);
                const mappedDirY = mapDHtoThree(dirYDH);
                const mappedDirZ = mapDHtoThree(dirZDH);

                // Disegna Componenti
                drawLink(prevThreePos, mappedOrigin);
                
                const jointSphere = new THREE.Mesh(jointGeom, jointMat);
                jointSphere.position.copy(mappedOrigin);
                robotGroup.add(jointSphere);

                drawReferenceFrame(mappedOrigin, mappedDirX, mappedDirY, mappedDirZ, colors);
                
                prevThreePos.copy(mappedOrigin);
            });
        }

        // --- INTERFACCIA UTENTE ---
        const jointsContainer = document.getElementById('joints-container');

        function renderUI() {
            jointsContainer.innerHTML = '';
            
            dhParams.forEach((param, index) => {
                const isR = param.type === 'R';
                const varLabel = isR ? 'θ (deg)' : 'd';
                const varValue = isR ? param.theta : param.d;
                const min = isR ? -180 : -10;
                const max = isR ? 180 : 10;
                const step = isR ? 1 : 0.1;

                const card = document.createElement('div');
                card.className = 'bg-gray-700/50 border border-gray-600 rounded p-3 relative text-sm';
                card.innerHTML = `
                    <div class="flex justify-between items-center mb-2">
                        <div class="flex items-center gap-2">
                            <span class="font-bold text-blue-400 bg-blue-900/40 px-1.5 py-0.5 rounded text-xs">J${index + 1}</span>
                            <select class="type-select bg-gray-800 text-xs rounded border border-gray-600 px-1 py-1" data-index="${index}">
                                <option value="R" ${isR ? 'selected' : ''}>Rivoluzione</option>
                                <option value="P" ${!isR ? 'selected' : ''}>Prismatico</option>
                            </select>
                        </div>
                        <button class="remove-btn text-gray-400 hover:text-red-400 font-bold px-2 py-0.5" data-index="${index}">✕</button>
                    </div>

                    <!-- Slider Variabile Principale -->
                    <div class="mb-3 bg-gray-800/80 p-2 rounded">
                        <div class="flex justify-between text-xs mb-1">
                            <span class="text-gray-400">Var: <strong class="text-white">${varLabel}</strong></span>
                            <span class="font-mono text-blue-300" id="val-${index}">${Number(varValue).toFixed(1)}</span>
                        </div>
                        <input type="range" min="${min}" max="${max}" step="${step}" value="${varValue}" class="slider" data-index="${index}" data-field="${isR ? 'theta' : 'd'}">
                    </div>

                    <!-- Parametri Esatti DH -->
                    <div class="grid grid-cols-4 gap-1 text-xs text-center">
                        <div><div class="text-gray-400 mb-0.5">a</div><input type="number" step="any" value="${param.a}" class="p-input w-full bg-gray-800 border border-gray-600 rounded px-1 text-center font-mono" data-index="${index}" data-field="a"></div>
                        <div><div class="text-gray-400 mb-0.5">α</div><input type="number" step="any" value="${param.alpha}" class="p-input w-full bg-gray-800 border border-gray-600 rounded px-1 text-center font-mono" data-index="${index}" data-field="alpha"></div>
                        <div><div class="text-gray-400 mb-0.5">d</div><input type="number" step="any" value="${param.d}" class="p-input w-full bg-gray-800 border border-gray-600 rounded px-1 text-center font-mono" data-index="${index}" data-field="d"></div>
                        <div><div class="text-gray-400 mb-0.5">θ</div><input type="number" step="any" value="${param.theta}" class="p-input w-full bg-gray-800 border border-gray-600 rounded px-1 text-center font-mono" data-index="${index}" data-field="theta"></div>
                    </div>
                `;
                jointsContainer.appendChild(card);
            });

            // Event Listeners Pannello
            document.querySelectorAll('.slider').forEach(el => {
                el.addEventListener('input', e => {
                    const idx = e.target.dataset.index;
                    const field = e.target.dataset.field;
                    const val = parseFloat(e.target.value);
                    dhParams[idx][field] = val;
                    // Aggiorna anche l'input numerico esatto
                    document.querySelector(`.p-input[data-index="${idx}"][data-field="${field}"]`).value = val;
                    document.getElementById(`val-${idx}`).innerText = val.toFixed(1);
                    buildRobot();
                });
            });

            document.querySelectorAll('.p-input').forEach(el => {
                el.addEventListener('input', e => {
                    const idx = e.target.dataset.index;
                    const field = e.target.dataset.field;
                    dhParams[idx][field] = parseFloat(e.target.value) || 0;
                    
                    // Se l'input corrisponde alla variabile dello slider, aggiorna lo slider e la label
                    const isVar = (dhParams[idx].type === 'R' && field === 'theta') || (dhParams[idx].type === 'P' && field === 'd');
                    if(isVar) {
                        const slider = document.querySelector(`.slider[data-index="${idx}"]`);
                        slider.value = dhParams[idx][field];
                        document.getElementById(`val-${idx}`).innerText = Number(dhParams[idx][field]).toFixed(1);
                    }
                    buildRobot();
                });
            });

            document.querySelectorAll('.type-select').forEach(el => {
                el.addEventListener('change', e => {
                    dhParams[e.target.dataset.index].type = e.target.value;
                    renderUI(); // Ri-renderizza per aggiornare i limiti dello slider
                    buildRobot();
                });
            });

            document.querySelectorAll('.remove-btn').forEach(el => {
                el.addEventListener('click', e => {
                    dhParams.splice(e.currentTarget.dataset.index, 1);
                    renderUI();
                    buildRobot();
                });
            });
        }

        document.getElementById('add-joint-btn').addEventListener('click', () => {
            dhParams.push({ type: 'R', a: 2, alpha: 0, d: 0, theta: 0 });
            renderUI();
            jointsContainer.scrollTop = jointsContainer.scrollHeight;
            buildRobot();
        });

        // Cambia solo i colori visivi degli assi, NON ruota telecamera o mondo
        document.getElementById('up-axis-select').addEventListener('change', buildRobot);

        // --- RENDER LOOP & RESIZE ---
        window.addEventListener('resize', () => {
            camera.aspect = container.clientWidth / container.clientHeight;
            camera.updateProjectionMatrix();
            renderer.setSize(container.clientWidth, container.clientHeight);
        });

        function animate() {
            requestAnimationFrame(animate);
            controls.update();
            renderer.render(scene, camera);
        }

        // --- START ---
        renderUI();
        buildRobot();
        animate();
        setTimeout(() => window.dispatchEvent(new Event('resize')), 100);
    </script>
</body>
